Statistical independence test with Yates correction

from scipy.stats import chi2_contingency import numpy as np table = np.array([[10, 20], [20, 10]]) print("===== INPUT DATA =====") print("Observed Table:") print(table, "\n") print("===== HYPOTHESES =====") print("Null Hypothesis (H0): There is NO association between Treatment and Success.") print(" Treatment does NOT affect outcome; variables are independent.") print("Alternative Hypothesis (H1): There IS an association between Treatment and Success.") print(" Treatment DOES affect outcome; variables are not independent.\n") chi_no_yates, p_no_yates, dof_no, expected_no = chi2_contingency(table, correction=False) chi_yates, p_yates, dof_y, expected_y = chi2_contingency(table, correction=True) print("===== EXPECTED VALUES UNDER H0 =====") print("Expected Table (if independence is assumed):") print(expected_no, "\n") print("===== CHI-SQUARE WITHOUT YATES CORRECTION =====") print(f"Chi-square value: {chi_no_yates:.4f}") print(f"P-value: {p_no_yates:.4f}") print("Meaning: This version may exaggerate significance when sample size or expected counts are small.\n") print("===== CHI-SQUARE WITH YATES CORRECTION =====") print(f"Chi-square value (Yates): {chi_yates:.4f}") print(f"P-value (Yates): {p_yates:.4f}") print("Meaning: Yates correction adjusts for discrete counts to reduce false significance.\n") print("===== RESULTS COMPARISON =====") print(f"Difference in Chi-square: {chi_no_yates - chi_yates:.4f}") if chi_yates < chi_no_yates: print("Yates correction reduces the chi-square value -> more conservative test.") print("It helps avoid false positives (Type-I errors), especially in a 2×2 table like this.\n") print("===== DECISION =====") alpha = 0.05 if p_yates < alpha: print(f"Since p = {p_yates:.4f} < 0.05, we REJECT the Null Hypothesis (H0).") else: print(f"Since p = {p_yates:.4f} >= 0.05, we FAIL to reject the Null Hypothesis (H0).") print("\n===== FINAL INTERPRETATION =====") print("Both tests show: Treatment and Success are not independent (p < 0.05).") print("But the Yates-corrected value is more trustworthy for smaller samples or 2×2 tables.")
